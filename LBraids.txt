    if(!u){alert('Pogrešni podaci'); return;}
    afterAuth(u); loginOverlay.classList.add('hidden');
  }

  // ---------- GALLERY FILTER ----------
  const gallery = document.getElementById('gallery');
  const filters = { length:'', thickness:'', color:'' };
  document.getElementById('lengthFilter').addEventListener('change', e=>{ filters.length=e.target.value; filterGallery(); });
  document.getElementById('thicknessFilter').addEventListener('change', e=>{ filters.thickness=e.target.value; filterGallery(); });
  document.getElementById('colorFilter').addEventListener('change', e=>{ filters.color=e.target.value; filterGallery(); });
  function filterGallery(){ gallery.querySelectorAll('.card').forEach(card=>{ const matchLength=!filters.length||card.dataset.length===filters.length; const matchThickness=!filters.thickness||card.dataset.thickness===filters.thickness; const matchColor=!filters.color||card.dataset.color===filters.color; card.style.display=(matchLength&&matchThickness&&matchColor)?'block':'none'; }); }

  // ---------- CALENDAR ----------
  let currentMonth = new Date().getMonth();
  let currentYear = new Date().getFullYear();
  const monthNames = ['Januar','Februar','Mart','April','Maj','Juni','Juli','August','Septembar','Oktobar','Novembar','Decembar'];
  
  function renderCalendar(){
    const calendarEl = document.getElementById('calendar');
    const monthYearEl = document.getElementById('monthYear');
    calendarEl.innerHTML='';
    monthYearEl.innerText = `${monthNames[currentMonth]} ${currentYear}`;
    
    const firstDay = new Date(currentYear, currentMonth, 1).getDay();
    const adjustedFirstDay = firstDay === 0 ? 6 : firstDay - 1;
    const daysInMonth = new Date(currentYear, currentMonth + 1, 0).getDate();
    
    for(let i=0; i<adjustedFirstDay; i++) {
      const emptyDiv = document.createElement('div');
      emptyDiv.className = 'day empty';
      calendarEl.appendChild(emptyDiv);
    }
    
    for(let d=1; d<=daysInMonth; d++){
      const date = new Date(currentYear, currentMonth, d);
      const iso = date.toISOString().split('T')[0];
      const div = document.createElement('div');
      div.innerText = d;
      div.className = 'day';
      
      let hasReservation = appointments[iso] && appointments[iso].some(a=>a.status==='confirmed');
      div.classList.add(hasReservation ? 'reserved' : 'available');
      div.onclick = ()=>selectDate(iso);
      calendarEl.appendChild(div);
    }
  }
  
  function changeMonth(direction){
    currentMonth += direction;
    if(currentMonth > 11){ currentMonth = 0; currentYear++; }
    if(currentMonth < 0){ currentMonth = 11; currentYear--; }
    renderCalendar();
  }
  
  function selectDate(date){
    if(currentUser.role==='admin'){ alert('Admin može uređivati termine direktno.'); return; }
    document.querySelectorAll('.calendar-grid .day').forEach(d=>d.classList.remove('selected'));
    event.target.classList.add('selected');
    timesEl.style.display='flex'; 
    timesEl.dataset.date=date;
  }
  
  const timesEl = document.getElementById('times');
  function reserve(time){
    const date = timesEl.dataset.date;
    if(!appointments[date]) appointments[date]=[];
    appointments[date].push({user:currentUser.name,email:currentUser.email,time,status:'pending'});
    localStorage.setItem('appointments',JSON.stringify(appointments));
    alert(`Termin rezervisan: ${date} u ${time}, čeka potvrdu admina.`);
    timesEl.style.display='none'; renderCalendar(); updateProfileUI();
  }

  // ---------- CHAT ----------
  const messagesEl = document.getElementById('messages');
  
  function sendMessage(){
    const text=document.getElementById('chatText').value;
    const imgFile=document.getElementById('chatImage').files[0];
    if(!text&&!imgFile) return alert('Unesi tekst ili sliku.');
    
    let msg={from:currentUser.role==='admin'?'admin':'user', text:'', img:'', timestamp:Date.now()};
    if(text) msg.text=text;
    
    if(imgFile){ 
      const reader=new FileReader(); 
      reader.onload=function(e){ 
        msg.img=e.target.result; 
        addMessage(msg); 
        saveChat(msg); 
      }; 
      reader.readAsDataURL(imgFile); 
    } else { 
      addMessage(msg); 
      saveChat(msg); 
    }
    document.getElementById('chatText').value=''; 
    document.getElementById('chatImage').value='';
  }
  
  function addMessage(msg){ 
    const div=document.createElement('div'); 
    div.className='message '+msg.from; 
    div.innerHTML=msg.text?msg.text:''; 
    if(msg.img) div.innerHTML+='<br><img src="'+msg.img+'" style="max-width:200px;border-radius:10px;margin-top:8px;">'; 
    messagesEl.appendChild(div); 
    messagesEl.scrollTop=messagesEl.scrollHeight; 
  }
  
  function saveChat(msg){ 
    const chatKey = currentUser.role === 'admin' ? currentChatUser : currentUser.email;
    if(!chats[chatKey]) chats[chatKey]=[]; 
    chats[chatKey].push(msg); 
    localStorage.setItem('chats',JSON.stringify(chats));
    if(currentUser.role==='admin') loadAdminChats();
  }
  
  function loadChat(){ 
    messagesEl.innerHTML=''; 
    const chatKey = currentUser.role === 'admin' ? currentChatUser : currentUser.email;
    if(chats[chatKey]) chats[chatKey].forEach(addMessage);
    
    if(currentUser.role === 'admin' && currentChatUser){
      const user = users.find(u=>u.email === currentChatUser);
      document.getElementById('chatHeader').innerText = user ? user.name : 'Chat';
    }
  }
  
  function loadAdminChats(){
    const chatUsersList = document.getElementById('chatUsersList');
    chatUsersList.innerHTML = '';
    
    // Prikaži sve usere koji imaju chat
    Object.keys(chats).forEach(email => {
      const user = users.find(u => u.email === email);
      if(!user) return;
      
      const lastMsg = chats[email][chats[email].length - 1];
      const div = document.createElement('div');
      div.className = 'chat-user-item';
      if(currentChatUser === email) div.classList.add('active');
      
      div.innerHTML = `
        <div class="name">${user.name}</div>
        <div class="preview">${lastMsg.text ? lastMsg.text.substring(0,30)+'...' : '📷 Slika'}</div>
      `;
      div.onclick = () => {
        currentChatUser = email;
        loadAdminChats();
        loadChat();
      };
      chatUsersList.appendChild(div);
    });
    
    // Ako nema selektovanog chata, selektuj prvog
    if(!currentChatUser && Object.keys(chats).length > 0){
      currentChatUser = Object.keys(chats)[0];
      loadChat();
    }
  }

  // ---------- PROFILE ----------
  function updateProfileUI(){
    document.getElementById('pName').innerText=currentUser.name;
    document.getElementById('pEmail').innerText=currentUser.email;
    let hair=''; 
    if(currentUser.length) hair=`Dužina: ${currentUser.length}, Boja: ${currentUser.color}, Tip: ${currentUser.type}`;
    document.getElementById('hairDetails').innerText=hair;
    let resDiv=document.getElementById('userReservations'); 
    resDiv.innerHTML='';
    
    for(let d in appointments){
      appointments[d].forEach((a, index)=>{
        if(currentUser.role==='admin'||a.email===currentUser.email){
          const div=document.createElement('div'); 
          div.className='reservation-item';
          
          const info = document.createElement('div');
          info.className = 'info';
          info.innerHTML = `<strong>${a.user}</strong><br>${d} u ${a.time}`;
          div.appendChild(info);
          
          const status = document.createElement('span');
          status.className = `status ${a.status}`;
          status.innerText = a.status === 'pending' ? 'Na čekanju' : 'Potvrđeno';
          div.appendChild(status);
          
          // Admin dugmad za potvrdu/otkazivanje
          if(currentUser.role === 'admin'){
            const actions = document.createElement('div');
            actions.className = 'actions';
            
            if(a.status === 'pending'){
              const confirmBtn = document.createElement('button');
              confirmBtn.className = 'confirm-btn';
              confirmBtn.innerText = 'Potvrdi';
              confirmBtn.onclick = () => confirmReservation(d, index);
              actions.appendChild(confirmBtn);
            }
            
            const cancelBtn = document.createElement('button');
            cancelBtn.className = 'cancel-btn';
            cancelBtn.innerText = 'Otkaži';
            cancelBtn.onclick = () => cancelReservation(d, index);
            actions.appendChild(cancelBtn);
            
            div.appendChild(actions);
          }
          
          resDiv.appendChild(div);
        }
      });
    }
  }
  
  function confirmReservation(date, index){
    if(confirm('Da li želiš potvrditi ovaj termin?')){
      appointments[date][index].status = 'confirmed';
      localStorage.setItem('appointments', JSON.stringify(appointments));
      updateProfileUI();
      renderCalendar();
      alert('Termin je uspješno potvrđen!');
    }
  }
  
  function cancelReservation(date, index){
    if(confirm('Da li želiš otkazati ovaj termin?')){
      appointments[date].splice(index, 1);
      if(appointments[date].length === 0) delete appointments[date];
      localStorage.setItem('appointments', JSON.stringify(appointments));
      updateProfileUI();
      renderCalendar();
      alert('Termin je otkazan.');
    }
  }
